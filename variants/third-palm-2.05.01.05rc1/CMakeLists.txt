cmake_minimum_required(VERSION 3.28)
project(SuperJumpThirdPalm LANGUAGES CXX)

set_property(SOURCE src/SessionLog.cpp src/SuperJump.rc APPEND PROPERTY
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/ModVersion.h")

add_executable(inspect-sites src/SiteResolver.cpp tools/inspect_sites.cpp)
target_compile_features(inspect-sites PRIVATE cxx_std_23)
target_compile_options(inspect-sites PRIVATE /W4 /WX /permissive-)
set_target_properties(inspect-sites PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded" CXX_SCAN_FOR_MODULES OFF)

include(CTest)

    set_target_properties(boost-config-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded" CXX_SCAN_FOR_MODULES OFF)

    set(p26_default "${CMAKE_CURRENT_SOURCE_DIR}/reference/P26")
endif()
set(SUPERJUMP_P26_REFERENCE "${p26_default}" CACHE PATH "Verified P26 contracts and dependency sources")
if(SUPERJUMP_P26_REFERENCE)
    enable_language(C ASM_MASM)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
    add_subdirectory("${SUPERJUMP_P26_REFERENCE}" p26-dependencies EXCLUDE_FROM_ALL)
    add_library(publisher-fixture-asm OBJECT tests/PublisherFixture.asm)
    add_executable(publisher-fixture tests/PublisherFixture.cpp $<TARGET_OBJECTS:publisher-fixture-asm>
        src/GameImage.cpp src/HookGroup.cpp src/SiteResolver.cpp)
    target_compile_features(publisher-fixture PRIVATE cxx_std_23)
    target_compile_options(publisher-fixture PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive->)
    target_link_libraries(publisher-fixture PRIVATE safetyhook)
    set_target_properties(publisher-fixture PROPERTIES CXX_SCAN_FOR_MODULES OFF)
endif()

if(SUPERJUMP_P26_REFERENCE)
    enable_language(RC)
    add_library(SuperJump_ThirdPalm SHARED
        src/Runtime.cpp src/BoostConfig.cpp src/GameImage.cpp src/HookGroup.cpp src/SiteResolver.cpp src/SessionLog.cpp src/SuperJump.rc)
    target_compile_features(SuperJump_ThirdPalm PRIVATE cxx_std_23)
    target_compile_options(SuperJump_ThirdPalm PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive->)
    target_compile_definitions(SuperJump_ThirdPalm PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(SuperJump_ThirdPalm SYSTEM PRIVATE "${SUPERJUMP_P26_REFERENCE}/source")
    target_link_libraries(SuperJump_ThirdPalm PRIVATE safetyhook version user32)
    target_link_options(SuperJump_ThirdPalm PRIVATE /Brepro)
    set_target_properties(SuperJump_ThirdPalm PROPERTIES SUFFIX ".asi" CXX_SCAN_FOR_MODULES OFF
        MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
